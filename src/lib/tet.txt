const mongoose = require("mongoose");

// Define the schema for responseData
const responseDataSchema = new mongoose.Schema({
  overview: {
    type: String,
    required: true,
  },
  keyFeatures: {
    type: [String],
    required: true,
  },
  roadmap: {
    type: [String],
    required: true,
  },
  techStackSuggestions: {
    frontend: {
      type: String,
      required: true,
    },
    backend: {
      type: String,
      required: true,
    },
    database: {
      type: String,
      required: true,
    },
    cloudStorage: {
      type: String,
      required: true,
    },
  },
  monetizationStrategy: {
    type: String,
    required: true,
  },
  marketingStrategy: {
    type: String,
    required: true,
  },
});

// Define the Answer schema to include the responseData
const answerSchema = new mongoose.Schema({
  user: {
    type: mongoose.Schema.Types.ObjectId,
    ref: "User",
  },
  message: {
    inputMessage: {
      type: String,
      required: true,
      minlength: 10,
    },
    responseMessage: {
      type: responseDataSchema, // Embed responseDataSchema into responseMessage
      required: true,
    },
    createdAt: {
      type: Date,
      default: Date.now,
    },
  },
});

// Define Messages model based on the schema
const Answer = mongoose.model("Answer", answerSchema);

module.exports = Answer;
